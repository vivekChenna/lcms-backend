# Case Core Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `cases` module — submission, role-scoped listing/detail, assignment/acceptance/rejection, escalation, priority changes, closure, auto-routing, and live SLA-state derivation.

**Architecture:** A new Fastify module `src/modules/cases/` following the existing feature-module pattern (entities/schemas/service/routes). Workflow actions (assign/accept/reject/escalate/close) live in `cases.workflow.service.ts`; submission/read/edit live in `cases.service.ts`. Pure logic (state machine, reference-number generation, SLA math, auto-routing matching) is split into standalone files with no DB dependency so it's unit-testable in isolation.

**Tech Stack:** Fastify 5, TypeORM, Zod 4 + fastify-type-provider-zod, PostgreSQL, Vitest.

**Spec:** `docs/superpowers/specs/2026-10-01-case-core-design.md`

## Global Constraints

- Schema changes only via committed migrations — never `synchronize: true`.
- Status changes only through named service functions validated against the state-machine transition table (WF-13/VC-14) — no generic "update status" path.
- All RBAC/visibility enforced server-side in the service layer, never trusted from the client (AC-03).
- Every mutating function ends with a `logAuditEvent` call; before/after values include `id` and `referenceNumber` for case entries (convention fixed in the Phase 1 review pass).
- TDD: write the failing test before the implementation, for every task.
- Requester-submitted cases are restricted to the `Internal` category server-side (VC-03), not just hidden in a UI.
- `closeCase`'s open-information-request guard, `waitingOnReason` population, document attachment at closure, and all notification dispatch are explicitly deferred to later Phase 2 sub-projects per the spec's §8 — do not add them here.

## Review Focus

- A requester submitting with a non-`Internal` category must be rejected server-side (VC-03), even if the client sends a valid category id belonging to Najiz/Baladi/HR Termination.
- `caseTypeId` not belonging to the submitted `categoryId`, or `businessUnitId` not belonging to the submitted `entityId`, must be rejected (VC-04/VC-05) — existence alone isn't enough, the relationship must hold.
- Reference numbers must never collide under concurrent submissions — the generator must use a DB sequence (`nextval`), not a count-then-insert race.
- A `legal_rep` can list/read every case (AC-05) but must get 403 attempting to accept/reject/escalate/close a case not assigned to and accepted by them — read-open, write-scoped is easy to under-enforce.
- Workflow actions attempted from the wrong status (closing a `Submitted` case, accepting a case that isn't `Assigned`) must be rejected via the state machine, not silently applied.

```typescript
// src/modules/cases/state-machine.ts
import { ConflictError } from '../../shared/errors';
import type { CaseStatus } from './entities/case.entity';

// Reassignment (PR-23) can happen from Assigned, In Progress, or Waiting on
// Input back to Assigned — a legal_manager can override at any point except
// once the case is Closed. The Waiting on Input edges have no caller yet in
// this sub-project (see spec §3/§8); they're defined here so sub-project 2
// inherits a verified, complete table instead of re-deriving it.
const TRANSITIONS: Record<CaseStatus, CaseStatus[]> = {
  Submitted: ['Assigned'],
  Assigned: ['In Progress', 'Submitted', 'Assigned'],
  'In Progress': ['Waiting on Input', 'Closed', 'Assigned'],
  'Waiting on Input': ['In Progress', 'Assigned'],
  Closed: [],
};

export function canTransition(from: CaseStatus, to: CaseStatus): boolean {
  return TRANSITIONS[from].includes(to);
}

export function assertTransition(from: CaseStatus, to: CaseStatus): void {
  if (!canTransition(from, to)) {
    throw new ConflictError(`Cannot transition case from '${from}' to '${to}'`);
  }
}
```

```typescript
// src/modules/cases/sla.ts
import type { Case } from './entities/case.entity';

// BRD requires an "at-risk" state (AU-11/DR-08) but specifies no numeric
// rule. Percentage-of-window scales with each case type's own SLA length
// instead of one fixed hour count across very different SLA durations
// (brainstorming session decision, 2026-10-01).
export const SLA_AT_RISK_WINDOW_FRACTION = 0.2;

export function computeDueDates(
  submittedAt: Date,
  slaAckHours: number,
  slaResolutionHours: number,
): { ackDueAt: Date; resolutionDueAt: Date } {
  const HOUR_MS = 60 * 60 * 1000;
  return {
    ackDueAt: new Date(submittedAt.getTime() + slaAckHours * HOUR_MS),
    resolutionDueAt: new Date(submittedAt.getTime() + slaResolutionHours * HOUR_MS),
  };
}

export type SlaState = 'on_track' | 'at_risk' | 'breached' | 'met';

export function deriveSlaState(params: {
  createdAt: Date;
  resolutionDueAt: Date;
  status: string;
  closedAt: Date | null;
  now?: Date;
}): SlaState {
  const now = params.now ?? new Date();

  if (params.status === 'Closed') {
    return params.closedAt !== null && params.closedAt.getTime() <= params.resolutionDueAt.getTime() ? 'met' : 'breached';
  }
  if (now.getTime() > params.resolutionDueAt.getTime()) {
    return 'breached';
  }
  const windowMs = params.resolutionDueAt.getTime() - params.createdAt.getTime();
  const remainingMs = params.resolutionDueAt.getTime() - now.getTime();
  if (windowMs > 0 && remainingMs <= windowMs * SLA_AT_RISK_WINDOW_FRACTION) {
    return 'at_risk';
  }
  return 'on_track';
}

export function deriveCaseAgeHours(createdAt: Date, closedAt: Date | null, now: Date = new Date()): number {
  const end = closedAt ?? now;
  return (end.getTime() - createdAt.getTime()) / (60 * 60 * 1000);
}

export function toCaseWithSla<T extends Case>(caseRow: T, now: Date = new Date()): T & { slaState: SlaState; caseAgeHours: number } {
  return {
    ...caseRow,
    slaState: deriveSlaState({
      createdAt: caseRow.createdAt,
      resolutionDueAt: caseRow.resolutionDueAt,
      status: caseRow.status,
      closedAt: caseRow.closedAt,
      now,
    }),
    caseAgeHours: deriveCaseAgeHours(caseRow.createdAt, caseRow.closedAt, now),
  };
}
```

```typescript
// src/modules/cases/auto-routing.ts
import type { DataSource } from 'typeorm';
import { RoutingRule } from '../master-data/entities/routing-rule.entity';

export type CaseMatchAttrs = {
  categoryId: string;
  caseTypeId: string;
  entityId: string;
  businessUnitId: string | null;
  departmentId: string;
  priority: string;
};

export function matchesCriteria(matchCriteria: Record<string, unknown>, caseAttrs: CaseMatchAttrs): boolean {
  return Object.entries(matchCriteria).every(([key, value]) => {
    if (value === undefined) {
      return true;
    }
    return (caseAttrs as Record<string, unknown>)[key] === value;
  });
}

export async function evaluateAutoRouting(dataSource: DataSource, caseAttrs: CaseMatchAttrs): Promise<RoutingRule | null> {
  const rules = await dataSource
    .getRepository(RoutingRule)
    .find({ where: { isActive: true }, order: { priorityOrder: 'ASC' } });
  return rules.find((rule) => matchesCriteria(rule.matchCriteria, caseAttrs)) ?? null;
}
```

```typescript
// src/modules/cases/cases.workflow.service.ts
import type { DataSource } from 'typeorm';
import { Case, CaseStatus } from './entities/case.entity';
import { assertTransition } from './state-machine';
import { NotFoundError, ForbiddenError } from '../../shared/errors';
import { logAuditEvent } from '../audit/audit.service';
import { assertValidLegalRep } from '../master-data/routing-rules.service';

export async function loadCaseOrThrow(dataSource: DataSource, caseId: string): Promise<Case> {
  const caseRow = await dataSource.getRepository(Case).findOne({ where: { id: caseId } });
  if (!caseRow) {
    throw new NotFoundError('Case not found');
  }
  return caseRow;
}

export async function assignCase(
  dataSource: DataSource,
  actorId: string,
  caseId: string,
  targetLegalRepId: string,
  auditAction: string = 'case.assigned',
): Promise<Case> {
  const caseRow = await loadCaseOrThrow(dataSource, caseId);
  assertTransition(caseRow.status as CaseStatus, 'Assigned');
  await assertValidLegalRep(dataSource, targetLegalRepId);

  const before = { id: caseRow.id, status: caseRow.status, assignedLegalRepId: caseRow.assignedLegalRepId };
  caseRow.status = 'Assigned';
  caseRow.assignedLegalRepId = targetLegalRepId;
  caseRow.assignmentAcceptedAt = null;

  const repo = dataSource.getRepository(Case);
  await repo.save(caseRow);
  await logAuditEvent(dataSource, {
    actorId,
    caseId: caseRow.id,
    action: auditAction,
    beforeValue: before,
    afterValue: {
      id: caseRow.id,
      status: caseRow.status,
      assignedLegalRepId: caseRow.assignedLegalRepId,
      referenceNumber: caseRow.referenceNumber,
    },
  });
  return caseRow;
}

export async function acceptAssignment(dataSource: DataSource, actorId: string, caseId: string): Promise<Case> {
  const caseRow = await loadCaseOrThrow(dataSource, caseId);
  if (caseRow.assignedLegalRepId !== actorId) {
    throw new ForbiddenError('Only the assigned legal representative can accept this case');
  }
  assertTransition(caseRow.status as CaseStatus, 'In Progress');

  const before = { id: caseRow.id, status: caseRow.status };
  caseRow.status = 'In Progress';
  caseRow.assignmentAcceptedAt = new Date();
  await dataSource.getRepository(Case).save(caseRow);
  await logAuditEvent(dataSource, {
    actorId,
    caseId: caseRow.id,
    action: 'case.accepted',
    beforeValue: before,
    afterValue: { id: caseRow.id, status: caseRow.status, referenceNumber: caseRow.referenceNumber },
  });
  return caseRow;
}

export async function rejectAssignment(dataSource: DataSource, actorId: string, caseId: string, reason: string): Promise<Case> {
  const caseRow = await loadCaseOrThrow(dataSource, caseId);
  if (caseRow.assignedLegalRepId !== actorId) {
    throw new ForbiddenError('Only the assigned legal representative can reject this case');
  }
  assertTransition(caseRow.status as CaseStatus, 'Submitted');

  const before = { id: caseRow.id, status: caseRow.status, assignedLegalRepId: caseRow.assignedLegalRepId };
  caseRow.status = 'Submitted';
  caseRow.assignedLegalRepId = null;
  caseRow.assignmentAcceptedAt = null;
  await dataSource.getRepository(Case).save(caseRow);
  await logAuditEvent(dataSource, {
    actorId,
    caseId: caseRow.id,
    action: 'case.rejected',
    beforeValue: before,
    afterValue: { id: caseRow.id, status: caseRow.status, reason, referenceNumber: caseRow.referenceNumber },
  });
  return caseRow;
}
```

```typescript
// src/modules/cases/cases.routes.ts
import type { FastifyInstance } from 'fastify';
import type { ZodTypeProvider } from 'fastify-type-provider-zod';
import { requireRole } from '../../shared/rbac';
import { toCaseWithSla } from './sla';
import { assignCase } from './cases.workflow.service';
import { assignCaseBodySchema, caseIdParamsSchema, caseResponseSchema } from './cases.schemas';

export default async function casesRoutes(app: FastifyInstance): Promise<void> {
  const typedApp = app.withTypeProvider<ZodTypeProvider>();

  typedApp.patch('/cases/:id/assign', {
    preHandler: [app.authenticate, requireRole(['legal_manager', 'admin'])],
    schema: {
      tags: ['Cases'],
      security: [{ bearerAuth: [] }],
      params: caseIdParamsSchema,
      body: assignCaseBodySchema,
      response: { 200: caseResponseSchema },
    },
  }, async (request) => {
    const caseRow = await assignCase(app.dataSource, request.user.id, request.params.id, request.body.targetLegalRepId);
    return toCaseWithSla(caseRow);
  });
}
```

```typescript
// src/modules/cases/cases.service.ts
import type { DataSource } from 'typeorm';
import { Case } from './entities/case.entity';
import { CaseCategory } from '../master-data/entities/case-category.entity';
import { CaseType } from '../master-data/entities/case-type.entity';
import { Department } from '../master-data/entities/department.entity';
import { OrgEntity } from '../master-data/entities/org-entity.entity';
import { BusinessUnit } from '../master-data/entities/business-unit.entity';
import { ValidationError } from '../../shared/errors';
import { logAuditEvent } from '../audit/audit.service';
import { computeDueDates } from './sla';
import { generateReferenceNumber } from './reference-number';
import { evaluateAutoRouting } from './auto-routing';
import { assignCase } from './cases.workflow.service';

export type CreateCaseInput = {
  title: string;
  description: string;
  declarationConfirmed: true;
  categoryId: string;
  caseTypeId: string;
  departmentId: string;
  entityId: string;
  businessUnitId?: string;
  priority: 'P1' | 'P2' | 'P3' | 'P4';
};

export type Actor = { id: string; role: string; departmentId: string | null; entityId: string | null; businessUnitId: string | null };

export async function submitCase(dataSource: DataSource, actor: Actor, input: CreateCaseInput): Promise<Case> {
  const category = await dataSource.getRepository(CaseCategory).findOne({ where: { id: input.categoryId } });
  if (!category || !category.isActive) {
    throw new ValidationError('Unknown or inactive categoryId');
  }
  if (actor.role === 'requester' && category.name !== 'Internal') {
    throw new ValidationError('Requesters may only submit cases in the Internal category (VC-03)');
  }

  const caseType = await dataSource.getRepository(CaseType).findOne({ where: { id: input.caseTypeId } });
  if (!caseType || !caseType.isActive) {
    throw new ValidationError('Unknown or inactive caseTypeId');
  }
  if (caseType.categoryId !== input.categoryId) {
    throw new ValidationError('caseTypeId does not belong to the submitted categoryId (VC-04)');
  }

  const department = await dataSource.getRepository(Department).findOne({ where: { id: input.departmentId } });
  if (!department || !department.isActive) {
    throw new ValidationError('Unknown or inactive departmentId');
  }

  // Requester's entity/business unit are pre-filled from their profile and
  // non-editable (PR-02); other submitter roles provide them directly.
  let entityId = input.entityId;
  let businessUnitId = input.businessUnitId ?? null;
  if (actor.role === 'requester') {
    if (!actor.entityId) {
      throw new ValidationError('Requester profile has no entity assigned');
    }
    entityId = actor.entityId;
    businessUnitId = actor.businessUnitId;
  }

  const entity = await dataSource.getRepository(OrgEntity).findOne({ where: { id: entityId } });
  if (!entity || !entity.isActive) {
    throw new ValidationError('Unknown or inactive entityId');
  }

  if (businessUnitId) {
    const businessUnit = await dataSource.getRepository(BusinessUnit).findOne({ where: { id: businessUnitId } });
    if (!businessUnit || !businessUnit.isActive) {
      throw new ValidationError('Unknown or inactive businessUnitId');
    }
    if (businessUnit.entityId !== entityId) {
      throw new ValidationError('businessUnitId does not belong to the submitted entityId (VC-05)');
    }
  }

  const submittedAt = new Date();
  const { ackDueAt, resolutionDueAt } = computeDueDates(submittedAt, caseType.slaAckHours, caseType.slaResolutionHours);

  const [{ seq }] = await dataSource.query(`SELECT nextval('case_reference_seq') AS seq`);
  const referenceNumber = generateReferenceNumber(entity.code, caseType.name, Number(seq));

  const repo = dataSource.getRepository(Case);
  const caseRow = await repo.save(
    repo.create({
      referenceNumber,
      title: input.title,
      description: input.description,
      declarationConfirmed: input.declarationConfirmed,
      categoryId: input.categoryId,
      caseTypeId: input.caseTypeId,
      departmentId: input.departmentId,
      entityId,
      businessUnitId,
      priority: input.priority,
      status: 'Submitted',
      isEscalated: false,
      escalationReason: null,
      waitingOnReason: null,
      requesterId: actor.id,
      assignedLegalRepId: null,
      assignmentAcceptedAt: null,
      ackDueAt,
      resolutionDueAt,
      slaBreachNotifiedAt: null,
      closureSummary: null,
      closureOutcomeId: null,
      closedById: null,
      closedAt: null,
    }),
  );

  await logAuditEvent(dataSource, {
    actorId: actor.id,
    caseId: caseRow.id,
    action: 'case.submitted',
    afterValue: { id: caseRow.id, referenceNumber: caseRow.referenceNumber, status: caseRow.status },
  });

  const match = await evaluateAutoRouting(dataSource, {
    categoryId: caseRow.categoryId,
    caseTypeId: caseRow.caseTypeId,
    entityId: caseRow.entityId,
    businessUnitId: caseRow.businessUnitId,
    departmentId: caseRow.departmentId,
    priority: caseRow.priority,
  });
  if (match) {
    return assignCase(dataSource, actor.id, caseRow.id, match.targetLegalRepId, 'case.auto_assigned');
  }

  return caseRow;
}
```

```typescript
// src/modules/cases/cases.service.ts — add these imports
import { NotFoundError } from '../../shared/errors';
import { PaginationQuery, paginationToSkipTake } from '../../shared/pagination';
import { toCaseWithSla } from './sla';

// add at the end of the file
export type CaseListQuery = PaginationQuery & {
  status?: string;
  priority?: string;
  categoryId?: string;
  caseTypeId?: string;
  entityId?: string;
  businessUnitId?: string;
  departmentId?: string;
  assignedLegalRepId?: string;
  search?: string;
  slaState?: 'on_track' | 'at_risk' | 'breached' | 'met';
};

const SLA_STATE_CONDITIONS: Record<string, string> = {
  met: `"status" = 'Closed' AND "closedAt" <= "resolutionDueAt"`,
  breached: `(("status" != 'Closed' AND "resolutionDueAt" < NOW()) OR ("status" = 'Closed' AND "closedAt" > "resolutionDueAt"))`,
  at_risk: `"status" != 'Closed' AND "resolutionDueAt" >= NOW() AND ("resolutionDueAt" - NOW()) <= ("resolutionDueAt" - "createdAt") * 0.2`,
  on_track: `"status" != 'Closed' AND ("resolutionDueAt" - NOW()) > ("resolutionDueAt" - "createdAt") * 0.2`,
};

export async function listCases(
  dataSource: DataSource,
  actor: Actor,
  query: CaseListQuery,
): Promise<{ items: ReturnType<typeof toCaseWithSla>[]; total: number; page: number; pageSize: number }> {
  const { skip, take } = paginationToSkipTake(query);
  const qb = dataSource.getRepository(Case).createQueryBuilder('case');

  if (actor.role === 'requester') {
    qb.andWhere('case."departmentId" = :actorDepartmentId', { actorDepartmentId: actor.departmentId });
  }

  if (query.status) qb.andWhere('case.status = :status', { status: query.status });
  if (query.priority) qb.andWhere('case.priority = :priority', { priority: query.priority });
  if (query.categoryId) qb.andWhere('case."categoryId" = :categoryId', { categoryId: query.categoryId });
  if (query.caseTypeId) qb.andWhere('case."caseTypeId" = :caseTypeId', { caseTypeId: query.caseTypeId });
  if (query.entityId) qb.andWhere('case."entityId" = :entityId', { entityId: query.entityId });
  if (query.businessUnitId) qb.andWhere('case."businessUnitId" = :businessUnitId', { businessUnitId: query.businessUnitId });
  if (query.departmentId) qb.andWhere('case."departmentId" = :departmentId', { departmentId: query.departmentId });
  if (query.assignedLegalRepId) qb.andWhere('case."assignedLegalRepId" = :assignedLegalRepId', { assignedLegalRepId: query.assignedLegalRepId });
  if (query.search) {
    qb.andWhere('(case.title ILIKE :search OR case.description ILIKE :search OR case."referenceNumber" ILIKE :search)', {
      search: `%${query.search}%`,
    });
  }
  if (query.slaState) {
    qb.andWhere(SLA_STATE_CONDITIONS[query.slaState]);
  }

  const total = await qb.getCount();
  const rows = await qb.orderBy('case.createdAt', 'DESC').skip(skip).take(take).getMany();

  return { items: rows.map((row) => toCaseWithSla(row)), total, page: query.page, pageSize: query.pageSize };
}

export async function getCaseById(dataSource: DataSource, actor: Actor, caseId: string) {
  const caseRow = await dataSource.getRepository(Case).findOne({ where: { id: caseId } });
  if (!caseRow) {
    throw new NotFoundError('Case not found');
  }
  if (actor.role === 'requester' && caseRow.departmentId !== actor.departmentId) {
    throw new NotFoundError('Case not found');
  }
  return toCaseWithSla(caseRow);
}
```

```typescript
      isEscalated: caseRow.isEscalated,

  });
```

```typescript
export type UpdateCaseInput = {
  title?: string;
  description?: string;
  categoryId?: string;
  caseTypeId?: string;
  departmentId?: string;
};

export async function updateCase(dataSource: DataSource, actor: Actor, caseId: string, input: UpdateCaseInput): Promise<Case> {
  const repo = dataSource.getRepository(Case);
  const caseRow = await repo.findOne({ where: { id: caseId } });
  if (!caseRow) {
    throw new NotFoundError('Case not found');
  }
  if (caseRow.requesterId !== actor.id && actor.role !== 'admin') {
    throw new ForbiddenError('Only the submitter or an admin can edit this case');
  }
  if (caseRow.status !== 'Submitted') {
    throw new ConflictError('Case can only be edited while Submitted');
  }
  if (input.categoryId && input.categoryId !== caseRow.categoryId) {
    const category = await dataSource.getRepository(CaseCategory).findOne({ where: { id: input.categoryId } });
    if (!category || !category.isActive) {
      throw new ValidationError('Unknown or inactive categoryId');
    }
  }
  const effectiveCategoryId = input.categoryId ?? caseRow.categoryId;
  if (input.caseTypeId && input.caseTypeId !== caseRow.caseTypeId) {
    const caseType = await dataSource.getRepository(CaseType).findOne({ where: { id: input.caseTypeId } });
    if (!caseType || !caseType.isActive) {
      throw new ValidationError('Unknown or inactive caseTypeId');
    }
    if (caseType.categoryId !== effectiveCategoryId) {
      throw new ValidationError('caseTypeId does not belong to the submitted categoryId (VC-04)');
    }
  }
  if (input.departmentId) {

    }
  }

  const before = {
    id: caseRow.id,
    title: caseRow.title,
    description: caseRow.description,
    categoryId: caseRow.categoryId,
    caseTypeId: caseRow.caseTypeId,
    departmentId: caseRow.departmentId,
  };

  if (input.title !== undefined) caseRow.title = input.title;
  if (input.description !== undefined) caseRow.description = input.description;
  if (input.categoryId !== undefined) caseRow.categoryId = input.categoryId;
  if (input.caseTypeId !== undefined) caseRow.caseTypeId = input.caseTypeId;
  if (input.departmentId !== undefined) caseRow.departmentId = input.departmentId;

  await repo.save(caseRow);
  await logAuditEvent(dataSource, {
    actorId: actor.id,
    caseId: caseRow.id,
    action: 'case.updated',
    beforeValue: before,
    afterValue: {
      id: caseRow.id,
      title: caseRow.title,
      description: caseRow.description,
      categoryId: caseRow.categoryId,
      caseTypeId: caseRow.caseTypeId,
      departmentId: caseRow.departmentId,
      referenceNumber: caseRow.referenceNumber,
    },
  });
  return caseRow;
}
```

```typescript
// src/modules/cases/cases.service.ts — add at the end of the file
export async function updatePriority(
  dataSource: DataSource,
  actorId: string,
  caseId: string,
  priority: Case['priority'],
): Promise<Case> {
  const repo = dataSource.getRepository(Case);
  const caseRow = await repo.findOne({ where: { id: caseId } });
  if (!caseRow) {
    throw new NotFoundError('Case not found');
  }
  if (caseRow.status === 'Closed') {
    throw new ConflictError('Cannot change priority on a Closed case');
  }

  const before = { id: caseRow.id, priority: caseRow.priority };
  caseRow.priority = priority;
  await repo.save(caseRow);
  await logAuditEvent(dataSource, {
    actorId,
    caseId: caseRow.id,
    action: 'case.priority_changed',
    beforeValue: before,
    afterValue: { id: caseRow.id, priority: caseRow.priority, referenceNumber: caseRow.referenceNumber },
  });
  return caseRow;
}
```
